សហគមន៍អាចចូលរួមតាមរយៈបណ្តាញផ្សេងៗ៖

- កិច្ចប្រជុំពិគ្រោះយោបល់សហគមន៍
- ការចូលរួមក្នុងគណៈកម្មការគ្រប់គ្រងសាលារៀន
- វគ្គផ្តល់មតិកែលម្អ និងការស្ទង់មតិ
- ឱកាសស្ម័គ្រចិត្តក្នុងសកម្មភាពគម្រោង
- ការចូលរួមតាមអ៊ីនធឺណិតតាមរយៈគេហទំព័រគម្រោង
